```pythonthon
"""
Python-based scoring engine for deep analysis.
Replaces AI reasoning with deterministic calculations.
"""

class DeepAnalysisScorer:
    """Calculate scores, grades, and recommendations using Python."""

    def calculate_composite_score(
        self,
        fundamental_metrics: dict,
        technical_metrics: dict,
        risk_metrics: dict
    ) -> float:
        """
        Calculate composite score (0.0-1.0) using weighted formula.

        Weights:
        - Fundamental: 40%
        - Technical: 30%
        - Risk: 30%
        """
        fundamental_score = self._calculate_fundamental_score(fundamental_metrics)
        technical_score = self._calculate_technical_score(technical_metrics)
        risk_score = self._calculate_risk_score(risk_metrics)

        composite = (
            fundamental_score * 0.4 +
            technical_score * 0.3 +
            ((5.0 - risk_score) / 5.0) * 0.3
        )

        return max(0.0, min(1.0, composite))

    def assign_grade(self, composite_score: float) -> str:
        """Assign letter grade based on composite score."""
        if composite_score >= 0.90: return "A+"
        if composite_score >= 0.85: return "A"
        if composite_score >= 0.80: return "A-"
        if composite_score >= 0.75: return "B+"
        if composite_score >= 0.70: return "B"
        if composite_score >= 0.65: return "B-"
        if composite_score >= 0.60: return "C+"
        if composite_score >= 0.55: return "C"
        if composite_score >= 0.50: return "C-"
        if composite_score >= 0.45: return "D+"
        if composite_score >= 0.40: return "D"
        if composite_score >= 0.35: return "D-"
        return "F"

    def generate_recommendation(
        self,
        grade: str,
        composite_score: float,
        risk_score: float
    ) -> tuple[str, str]:
        """
        Generate BUY/HOLD/SELL recommendation with rationale.

        Returns:
            (recommendation, rationale)
        """
        # Simple deterministic logic
        if grade in ["A+", "A", "A-"] and risk_score <= 3.0:
            return "BUY", f"Strong fundamentals (grade {grade}) with manageable risk ({risk_score}/5)"

        if grade in ["B+", "B"] and risk_score <= 3.5:
            return "HOLD", f"Good quality (grade {grade}) but monitor risk ({risk_score}/5)"

        if grade in ["B-", "C+", "C"]:
            return "HOLD", f"Average quality (grade {grade}), consider alternatives"

        return "SELL", f"Underperforming (grade {grade}), high risk ({risk_score}/5)"

    def _calculate_fundamental_score(self, metrics: dict) -> float:
        """Calculate fundamental score from metrics."""
        score = 0.5  # Base score

        # ROE bonus
        roe = metrics.get("roe", 0)
        if roe > 20: score += 0.2
        elif roe > 15: score += 0.1

        # Debt penalty
        debt_equity = metrics.get("debt_to_equity", 0)
        if debt_equity > 0.5: score -= 0.1
        if debt_equity > 1.0: score -= 0.2

        # Growth bonus
        revenue_growth = metrics.get("revenue_growth", 0)
        if revenue_growth > 0.15: score += 0.2
        elif revenue_growth > 0.10: score += 0.1

        return max(0.0, min(1.0, score))

    def _calculate_technical_score(self, metrics: dict) -> float:
        """Calculate technical score from metrics."""
        score = 0.5  # Base score

        # Momentum indicators
        rsi = metrics.get("rsi", 50)
        if 40 <= rsi <= 60: score += 0.1  # Neutral zone
        elif rsi < 30: score -= 0.2  # Oversold
        elif rsi > 70: score -= 0.2  # Overbought

        # Trend
        sma_50 = metrics.get("sma_50", 0)
        sma_200 = metrics.get("sma_200", 0)
        current_price = metrics.get("current_price", 0)

        if sma_50 > sma_200 and current_price > sma_50:
            score += 0.3  # Strong uptrend
        elif sma_50 < sma_200 and current_price < sma_50:
            score -= 0.3  # Strong downtrend

        return max(0.0, min(1.0, score))

    def _calculate_risk_score(self, metrics: dict) -> float:
        """
        Calculate risk score (0-5 scale).

        0-1: Very Low/Low
        1-2: Low
        2-3: Moderate
        3-4: High
        4-5: Very High
        """
        volatility = metrics.get("volatility", 20)
        max_drawdown = abs(metrics.get("max_drawdown", 0))
        beta = metrics.get("beta", 1.0)

        # Base score from volatility
        score = (volatility / 35) * 2.0  # Normalize to 0-2 range

        # Drawdown penalty
        score += (max_drawdown / 50) * 1.5  # 0-1.5 range

        # Beta adjustment
        if beta > 1.5: score += 0.5
        elif beta < 0.5: score -= 0.3

        return max(0.0, min(5.0, score))

### Phase 2: Simplified Deep Analysis Crew

### Phase 4: Python Report Generator

**Create:** `src/finwiz/reporting/deep_analysis_report_generator.py`

```pythonthon
"""
Python-based report generator using Jinja2 templates.
Replaces AI HTML generation.
"""

from jinja2 import Environment, FileSystemLoader
from pathlib import Path

class DeepAnalysisReportGenerator:
    """Generate HTML reports using Jinja2 templates."""

    def __init__(self):
        template_dir = Path(__file__).parent.parent / "templates"
        self.env = Environment(loader=FileSystemLoader(template_dir))

    def generate_report(self, analysis_result: dict) -> str:
        """
        Generate HTML report from analysis result.

        Args:
            analysis_result: Dict with ticker, grade, scores, etc.

        Returns:
            HTML string
        """
        template = self.env.get_template("deep_analysis_report.html.j2")
        return template.render(**analysis_result)
